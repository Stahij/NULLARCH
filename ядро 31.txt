        diagEl.textContent = '';
        diagEl.style.display = 'none';
      }

      window.addEventListener('error', function (ev) {
        reportDiag('Runtime error: ' + (ev && ev.message ? ev.message : 'Script failure'));
      });
      window.addEventListener('unhandledrejection', function (ev) {
        var reason = ev && ev.reason ? (ev.reason.message || ev.reason) : 'Promise rejection';
        reportDiag('Async error: ' + reason);
      });

      function loadScriptSequential(urls, index, callback) {
        if (index >= urls.length) {
          callback(new Error('Failed to load script from all CDNs'));
          return;
        }
        const script = document.createElement('script');
        script.src = urls[index];
        script.async = false;
        script.onload = function () {
          callback(null);
        };
        script.onerror = function () {
          loadScriptSequential(urls, index + 1, callback);
        };
        document.head.appendChild(script);
      }

      loadScriptSequential(THREE_CDNS, 0, function (threeErr) {

      // 1. UNMODIFIED CENTRAL PLASMA CORE CLASS

      class PlasmaCoreOnly {
        constructor(scene, quality = "medium") {
          this.scene = scene;
          this.group = new THREE.Group();
          this.disposed = false;

          const low = quality === "low";
          const high = quality === "high";

          this.uniforms = {
            uTime: { value: 0 },
            uDetail: { value: low ? 1.0 : high ? 2.5 : 1.8 },
            uIntensity: { value: 1.0 },
            uPulse: { value: 0.032 },
            uHue: { value: 0.0 },
            uSurge: { value: 0.0 },
            uSurgeDir: {
              value: new THREE.Vector3(0.6, 0.7, 0.4).normalize()
            }
          };

          const U = this.uniforms;

          const plasmaField = `
            float gyroid(vec3 p) {
              return dot(sin(p), cos(p.yzx));
            }

            float plasmaField(vec3 p, float t, float detail) {
              vec3 q = p * 2.1 +
                vec3(t * 0.42, -t * 0.31, t * 0.27);

              float a = gyroid(q);

              vec3 w = p * 3.4 -
                vec3(t * 0.55, t * 0.38, -t * 0.44) +
                a * 0.65;

              float b = gyroid(w);
              float v = a * 0.58 + b * 0.42;

              if (detail > 1.5) {
                v += gyroid(
                  w * 2.15 + vec3(b * 0.9, -t * 0.7, t * 0.5)
                ) * 0.22 * clamp(detail - 1.5, 0.0, 1.0);
              }

              return v;
            }
          `;

          // Внутренняя плазменная сфера
          const coreVertex = `
            uniform float uTime;
            uniform float uPulse;
            uniform float uSurge;

            varying vec3 vN;
            varying vec3 vP;
            varying vec3 vW;

            void main() {
              vP = position;

              float a =
                sin(position.x * 3.2 + uTime * 1.6) *
                cos(position.y * 3.1 - uTime * 1.3);

              float b =
                sin(position.z * 4.1 + uTime * 2.1 +
                    position.x * 2.0);

              vec3 p = position +
                normal * ((a * 0.65 + b * 0.35) *
                (uPulse + uSurge * 0.025));

              vec4 w = modelMatrix * vec4(p, 1.0);

              vW = w.xyz;
              vN = normalize(mat3(modelMatrix) * normal);

              gl_Position = projectionMatrix * viewMatrix * w;
            }
          `;

          const coreFragment = `
            precision highp float;

            uniform float uTime;
            uniform float uDetail;
            uniform float uIntensity;
            uniform float uHue;
            uniform float uSurge;
            uniform vec3 uSurgeDir;

            varying vec3 vN;
            varying vec3 vP;
            varying vec3 vW;

            ${plasmaField}

            void main() {
              vec3 N = normalize(vN);
              vec3 V = normalize(cameraPosition - vW);

              float fres = pow(
                1.0 - max(dot(N, V), 0.0), 2.3
              );

              vec3 deep =
                vP * 1.15 -
                normalize(vP - cameraPosition * 0.25) * 0.32;

              float a = plasmaField(vP, uTime, uDetail);
              float b = plasmaField(
                deep, uTime * 0.72 + 1.7,
                max(1.0, uDetail - 0.6)
              );

              float f = pow(
                clamp(1.0 - abs(a * 0.68 + b * 0.32), 0.0, 1.0),
                3.0
              );

              float heat = smoothstep(
                -0.35, 0.85, b * 0.6 + a * 0.4
              );

              vec3 c = mix(
                vec3(0.012, 0.065, 0.32),
                vec3(0.02, 0.28, 0.85),
                heat
              );

              c = mix(
                c,
                vec3(0.04, 0.78 + uHue * 0.35, 1.0),
                smoothstep(0.22, 0.78, f)
              );

              c = mix(
                c,
                vec3(0.86, 0.97, 1.0),
                smoothstep(0.76, 0.98, f) * 0.92
              );

              float surge =
                (1.0 - smoothstep(
                  0.0, 0.55,
                  distance(normalize(vP), uSurgeDir)
                )) * uSurge;

              c += vec3(0.86, 0.97, 1.0) *
                surge * (0.55 + f * 1.15);

              c += vec3(0.04, 0.78, 1.0) * fres * 0.72;
              c += vec3(0.86, 0.97, 1.0) *
                pow(fres, 4.2) * 0.45;

              gl_FragColor = vec4(c * uIntensity, 1.0);
            }
          `;

          const core = new THREE.Mesh(
            new THREE.IcosahedronGeometry(1, low ? 3 : 4),
            new THREE.ShaderMaterial({
              uniforms: U,
              vertexShader: coreVertex,
              fragmentShader: coreFragment
            })
          );

          core.scale.setScalar(0.88);
          this.group.add(core);

          // Внешняя полупрозрачная энергетическая оболочка
          const haloVertex = `
            uniform float uTime;
            uniform float uPulse;

            varying vec3 vN;
            varying vec3 vP;
            varying vec3 vW;

            void main() {
              vP = position;

              float b = 1.0 +
                sin(uTime * 1.45 + position.y * 2.2) *
                (uPulse * 1.25);

              vec4 w = modelMatrix *
                vec4(position * b, 1.0);

              vW = w.xyz;
              vN = normalize(mat3(modelMatrix) * normal);

              gl_Position = projectionMatrix * viewMatrix * w;
            }
          `;

          const haloFragment = `
            precision mediump float;

            uniform float uTime;
            uniform float uDetail;
            uniform float uIntensity;
            uniform float uHue;
            uniform float uSurge;

            varying vec3 vN;
            varying vec3 vP;
            varying vec3 vW;

            ${plasmaField}

            void main() {
              vec3 N = normalize(vN);
              vec3 V = normalize(cameraPosition - vW);

              float nd = max(dot(N, V), 0.0);
              float rim = sin(nd * 3.14159265);

              float halo =
                pow(nd, 1.65) * 0.42 +
                pow(rim, 2.2) * 0.65;

              vec3 p = vP * 1.75;

              float a = uTime * 0.48 + p.y * 1.4;
              float s = sin(a);
              float c = cos(a);

              p.xz = mat2(c, -s, s, c) * p.xz;

              float stream = plasmaField(
                p, -uTime * 0.65,
                max(1.0, uDetail - 0.5)
              );

              float wisps = smoothstep(
                0.15, 0.88, 1.0 - abs(stream)
              );

                vec3(0.03, 0.68 + uHue * 0.3, 1.0),

              );

                halo * (0.24 + wisps * 0.44) *
                uIntensity * (1.0 + uSurge * 0.25);

              gl_FragColor = vec4(
                col, clamp(alpha, 0.0, 0.85)
              );
            }
          `;

          const halo = new THREE.Mesh(
            new THREE.IcosahedronGeometry(1.25, low ? 2 : 3),
            new THREE.ShaderMaterial({
              uniforms: U,
              vertexShader: haloVertex,
              fragmentShader: haloFragment,
              transparent: true,
              depthWrite: false,
              blending: THREE.AdditiveBlending
            })
          );

          halo.scale.setScalar(0.94);
          this.group.add(halo);

          this.scene.add(this.group);
        }

        update(time, state = "idle", speed = 1.0) {
          if (this.disposed) return;

          const u = this.uniforms;

          u.uTime.value = time * speed;

          u.uIntensity.value =
            state === "processing" ? 1.38 :
            state === "speaking" ? 1.48 :
            state === "listening" ? 1.24 : 1.0;

          u.uPulse.value =
            state === "speaking" ? 0.075 :
            state === "listening" ? 0.055 : 0.032;

          u.uHue.value =
            state === "processing" ? -0.12 :
            state === "listening" ? 0.14 : 0.0;

          u.uSurge.value = 0.5 + 0.5 * Math.sin(time * 0.65);

          u.uSurgeDir.value.set(
            Math.sin(time * 0.31),
            Math.cos(time * 0.27),
            Math.sin(time * 0.19)
          ).normalize();

          this.group.rotation.y = time * 0.14 * speed;
          this.group.rotation.x = Math.sin(time * 0.35) * 0.05;
        }

        dispose() {
          if (this.disposed) return;

          this.disposed = true;
          this.scene.remove(this.group);

          this.group.traverse(obj => {
            if (obj.geometry) obj.geometry.dispose();

            if (obj.material) {
              if (Array.isArray(obj.material)) {
                obj.material.forEach(m => m.dispose());
              } else {
                obj.material.dispose();
              }
            }
          });
        }
      }
