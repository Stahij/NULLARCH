     * ============================================================================
     * JARVIS LAUNCHER 3D AI CORE — MOBILE-FIRST GRAPHICS ENGINE
     * Target Hardware: Xiaomi Redmi Note 11 Pro 4G (2209116AG), Snapdragon 732G,
     * Adreno 618 GPU, Android 13 WebView (Chromium WebGL 2).
     *
     * Key Engineering Decisions:
     * 1. Zero post-processing / zero FBO render targets (saves ~45% Adreno 618
     *    memory bandwidth and tile-binning overhead).
     * 2. Exactly 4 Draw Calls per frame:
     *    - DC 1: Inner Plasma Core (gyroid + domain-warped volumetric plasma)
     *    - DC 2: Outer Volumetric Energy Sheath & Corona (analytical Fresnel)
     *    - DC 3: 5 Organic Plasma Ribbons (single merged BufferGeometry)
     *    - DC 4: 36 Orbital Energy Sparks (single GPU-animated Points batch)
     * 3. 100% GPU Vertex/Fragment Animation: zero per-frame CPU buffer uploads.
     * 4. Rolling Frame-Time Governor + 8s Idle 30FPS Power-Save + Lifecycle Suspend.
     * ============================================================================
     */

    // ---------------------------------------------------------------------------
    // 1. FAST MOBILE GLSL NOISE CHUNK (Gyroid + Trigonometric Domain Warping)
    // Replaces expensive 3D Simplex permutation loops with ALU-friendly trig folds
    // tuned for Adreno 618 scalar/vector pipelines.
    // ---------------------------------------------------------------------------
    const MOBILE_PLASMA_GLSL = /* glsl */ `
      // Fast 3D Gyroid field: smooth, organic, cellular/filamentous structure
      float gyroid(vec3 p) {
        return dot(sin(p), cos(p.yzx));
      }

      // Multi-layered domain-warped plasma field controlled by uDetail (1.0 .. 3.0)
      float plasmaField(vec3 p, float t, float detail) {
        vec3 q = p * 2.1 + vec3(t * 0.42, -t * 0.31, t * 0.27);
        float n1 = gyroid(q);

        // Domain warp pass 1
        vec3 warp = p * 3.4 - vec3(t * 0.55, t * 0.38, -t * 0.44) + n1 * 0.65;
        float n2 = gyroid(warp);

        float combined = n1 * 0.58 + n2 * 0.42;

        // Optional high-frequency filament detail (skipped on Low profile for GPU savings)
        if (detail > 1.5) {
          float n3 = gyroid(warp * 2.15 + vec3(n2 * 0.9, -t * 0.7, t * 0.5));
          combined += n3 * 0.22 * clamp(detail - 1.5, 0.0, 1.0);
        }

        return combined;
      }
    `;

    // ---------------------------------------------------------------------------
    // 5. DRAW CALL 1: INNER VOLUMETRIC PLASMA CORE
    // High-contrast deep cobalt core with glowing cyan/white gyroid plasma streams,
    // internal depth parallax, and localized energy surges.
    // ---------------------------------------------------------------------------
    const innerCoreGeo = new THREE.IcosahedronGeometry(1.0, 4); // 2,562 vertices — smooth & light on Adreno 618
    const innerCoreMat = new THREE.ShaderMaterial({
      uniforms: sharedUniforms,
      vertexShader: /* glsl */ `
        uniform float uTime;
        uniform float uPulseAmp;
        uniform float uSurge;

        varying vec3 vNormalW;
        varying vec3 vPosLocal;
        varying vec3 vWorldPos;
        varying float vDisplace;

        void main() {
          vPosLocal = position;

          // Lightweight GPU vertex displacement: organic non-spherical breathing
          float w1 = sin(position.x * 3.2 + uTime * 1.6) * cos(position.y * 3.1 - uTime * 1.3);
          float w2 = sin(position.z * 4.1 + uTime * 2.1 + position.x * 2.0);
          float disp = (w1 * 0.65 + w2 * 0.35) * (uPulseAmp + uSurge * 0.025);
          vDisplace = disp;

          vec3 displaced = position + normal * disp;
          vec4 worldPos = modelMatrix * vec4(displaced, 1.0);
          vWorldPos = worldPos.xyz;
          vNormalW = normalize(mat3(modelMatrix) * normal);

          gl_Position = projectionMatrix * viewMatrix * worldPos;
        }
      `,
      fragmentShader: /* glsl */ `
        precision highp float;

        uniform float uTime;
        uniform float uDetail;
        uniform float uIntensity;
        uniform float uHueShift;
        uniform float uSurge;
        uniform vec3 uSurgeDir;

        varying vec3 vNormalW;
        varying vec3 vPosLocal;
        varying vec3 vWorldPos;
        varying float vDisplace;

        ${MOBILE_PLASMA_GLSL}

        void main() {
          vec3 N = normalize(vNormalW);
          vec3 V = normalize(cameraPosition - vWorldPos);
          float NdotV = max(dot(N, V), 0.0);
          float fresnel = pow(1.0 - NdotV, 2.3);

          // Fake internal volumetric depth via view-ray refraction offset (parallax shell)
          vec3 viewLocal = normalize(vPosLocal - cameraPosition * 0.25);
          vec3 deepCoord = vPosLocal * 1.15 - viewLocal * 0.32;

          float outerStream = plasmaField(vPosLocal, uTime, uDetail);
          float innerStream = plasmaField(deepCoord, uTime * 0.72 + 1.7, max(1.0, uDetail - 0.6));

          // Ridge extraction for crisp energetic filaments
          float filament = 1.0 - abs(outerStream * 0.68 + innerStream * 0.32);
          filament = pow(clamp(filament, 0.0, 1.0), 3.0);

          float coreHeat = smoothstep(-0.35, 0.85, innerStream * 0.6 + outerStream * 0.4);

          // Color palette: Deep Cobalt -> Electric Blue -> Cyan -> Crisp White-Blue
          vec3 deepCobalt  = vec3(0.012, 0.065, 0.32);
          vec3 royalBlue   = vec3(0.02, 0.28, 0.85);
          vec3 electricCyan = vec3(0.04, 0.78 + uHueShift * 0.35, 1.0);
          vec3 whitePlasma = vec3(0.86, 0.97, 1.0);

          vec3 col = mix(deepCobalt, royalBlue, coreHeat);
          col = mix(col, electricCyan, smoothstep(0.22, 0.78, filament));
          col = mix(col, whitePlasma, smoothstep(0.76, 0.98, filament) * 0.92);

          // Localized stochastic energy surge / flare
          float surgeMask = smoothstep(0.55, 0.0, distance(normalize(vPosLocal), uSurgeDir)) * uSurge;
          col += whitePlasma * surgeMask * (0.55 + filament * 1.15);

          // Analytical Fresnel rim & internal core brightness
          col += electricCyan * fresnel * 0.72;
          col += whitePlasma * pow(fresnel, 4.2) * 0.45;
          col *= uIntensity;

          gl_FragColor = vec4(col, 1.0);
        }
      `
    });
    const innerCoreMesh = new THREE.Mesh(innerCoreGeo, innerCoreMat);
    coreRoot.add(innerCoreMesh);

    // ---------------------------------------------------------------------------
    // 6. DRAW CALL 2: TRANSLUCENT OUTER ENERGY SHELL & VOLUMETRIC CORONA HALO
    // Replaces expensive post-processing UnrealBloomPass with a single additive
    // analytical Fresnel + swirling filament shell!
    // ---------------------------------------------------------------------------
    const outerShellGeo = new THREE.IcosahedronGeometry(1.36, 3); // 642 vertices — ultra cheap
    const outerShellMat = new THREE.ShaderMaterial({
      transparent: true,
      depthWrite: false,
      blending: THREE.AdditiveBlending,
      side: THREE.FrontSide,
      uniforms: sharedUniforms,
      vertexShader: /* glsl */ `
        uniform float uTime;
        uniform float uPulseAmp;

        varying vec3 vNormalW;
        varying vec3 vWorldPos;
        varying vec3 vPosLocal;

        void main() {
          vPosLocal = position;
          float breathe = 1.0 + sin(uTime * 1.45 + position.y * 2.2) * (uPulseAmp * 1.25);
          vec4 worldPos = modelMatrix * vec4(position * breathe, 1.0);
          vWorldPos = worldPos.xyz;
          vNormalW = normalize(mat3(modelMatrix) * normal);

          gl_Position = projectionMatrix * viewMatrix * worldPos;
        }
      `,
      fragmentShader: /* glsl */ `
        precision mediump float;

        uniform float uTime;
        uniform float uDetail;
        uniform float uIntensity;
        uniform float uHueShift;
        uniform float uSurge;

        varying vec3 vNormalW;
        varying vec3 vWorldPos;
        varying vec3 vPosLocal;

        ${MOBILE_PLASMA_GLSL}

        void main() {
          vec3 N = normalize(vNormalW);
          vec3 V = normalize(cameraPosition - vWorldPos);
          float NdotV = max(dot(N, V), 0.0);

          // Bell-curve volumetric falloff: creates both inner energy sheath AND soft outer bloom halo
          float rimBand = sin(NdotV * 3.14159265);
          float softHalo = pow(NdotV, 1.65) * 0.42 + pow(rimBand, 2.2) * 0.65;

          // Swirling counter-rotating plasma tendrils
          vec3 p = vPosLocal * 1.75;
          float swirl = uTime * 0.48 + p.y * 1.4;
          float s = sin(swirl);
          float c = cos(swirl);
          p.xz = mat2(c, -s, s, c) * p.xz;

          float stream = plasmaField(p, -uTime * 0.65, max(1.0, uDetail - 0.5));
          float wisps = smoothstep(0.15, 0.88, 1.0 - abs(stream));

          vec3 outerCyan = vec3(0.03, 0.68 + uHueShift * 0.3, 1.0);
          vec3 innerAzure = vec3(0.02, 0.26, 0.88);
          vec3 col = mix(innerAzure, outerCyan, wisps);

          float alpha = softHalo * (0.24 + wisps * 0.44) * uIntensity * (1.0 + uSurge * 0.25);
          gl_FragColor = vec4(col, clamp(alpha, 0.0, 0.85));
        }
      `
    });
    const outerShellMesh = new THREE.Mesh(outerShellGeo, outerShellMat);
    coreRoot.add(outerShellMesh);

    // ---------------------------------------------------------------------------
    // 7. DRAW CALL 3: 5 ORGANIC PLASMA RIBBONS (SINGLE MERGED BUFFERGEOMETRY)
    // All 5 asymmetric twisted energy streams live in one static VBO/IBO and are
    // deformed smoothly in the vertex shader — zero per-frame CPU updates!
    // ---------------------------------------------------------------------------
    function createMergedRibbonsGeometry(ribbonCount = 5, lengthSegments = 72) {
      const vertsPerRibbon = (lengthSegments + 1) * 2;
      const totalVerts = ribbonCount * vertsPerRibbon;
      const indicesPerRibbon = lengthSegments * 6;
      const totalIndices = ribbonCount * indicesPerRibbon;

      const positions = new Float32Array(totalVerts * 3);
      const uvs = new Float32Array(totalVerts * 2);
      const ribbonMeta = new Float32Array(totalVerts * 3); // [ribbonIndex, seed, side (-1 or +1)]
      const indices = new Uint16Array(totalIndices);

      const qTilt = new THREE.Quaternion();
      const euler = new THREE.Euler();
      const pCenter = new THREE.Vector3();
      const pNext = new THREE.Vector3();
      const tangent = new THREE.Vector3();
      const radial = new THREE.Vector3();
      const sideVec = new THREE.Vector3();
      const vPos = new THREE.Vector3();

      let vOffset = 0;
      let iOffset = 0;

      for (let r = 0; r < ribbonCount; r++) {
        const seed = (r + 1) * 1.6180339;
        const baseRadius = 1.48 + (r % 3) * 0.16;
        const arcSpan = Math.PI * (1.2 + (r % 2) * 0.32);
        const maxHalfWidth = 0.045 + (r % 2) * 0.016;

        // Asymmetric non-planar orientation for each plasma stream
        euler.set(
          (r * 1.22) - 1.1,
          (r * 2.14) + 0.5,
          (r * 0.91) - 0.7
        );
        qTilt.setFromEuler(euler);

        const evalPoint = (u, target) => {
          const theta = (u - 0.5) * arcSpan + seed;
          const rad = baseRadius + Math.sin(theta * 2.0 + seed) * 0.16;
          const yWave = Math.sin(theta * 2.5 + seed * 1.4) * 0.34;
          target.set(Math.cos(theta) * rad, yWave, Math.sin(theta) * rad);
          target.applyQuaternion(qTilt);
        };

        const baseVertexIndex = vOffset;

        for (let i = 0; i <= lengthSegments; i++) {
          const u = i / lengthSegments;
          evalPoint(u, pCenter);
          evalPoint(Math.min(1.0, u + 0.01), pNext);

          tangent.subVectors(pNext, pCenter).normalize();
          radial.copy(pCenter).normalize();
          sideVec.crossVectors(tangent, radial).normalize();

          // Smooth tapering at both ends of the ribbon
          const taper = Math.pow(Math.sin(u * Math.PI), 0.7);
          const halfW = maxHalfWidth * taper;

          for (let s = 0; s < 2; s++) {
            const sideSign = s === 0 ? -1 : 1;
            vPos.copy(pCenter).addScaledVector(sideVec, halfW * sideSign);

            positions[vOffset * 3] = vPos.x;
            positions[vOffset * 3 + 1] = vPos.y;
            positions[vOffset * 3 + 2] = vPos.z;

            uvs[vOffset * 2] = u;
            uvs[vOffset * 2 + 1] = s;

            ribbonMeta[vOffset * 3] = r / ribbonCount;
            ribbonMeta[vOffset * 3 + 1] = seed;
            ribbonMeta[vOffset * 3 + 2] = sideSign;

            vOffset++;
          }
        }

        for (let i = 0; i < lengthSegments; i++) {
          const a = baseVertexIndex + i * 2;
          const b = a + 1;
          const c = a + 2;
          const d = a + 3;
          indices[iOffset++] = a;
          indices[iOffset++] = c;
          indices[iOffset++] = b;
          indices[iOffset++] = b;
          indices[iOffset++] = c;
          indices[iOffset++] = d;
        }
      }

      const geo = new THREE.BufferGeometry();
      geo.setAttribute('position', new THREE.BufferAttribute(positions, 3));
      geo.setAttribute('uv', new THREE.BufferAttribute(uvs, 2));
      geo.setAttribute('aRibbonMeta', new THREE.BufferAttribute(ribbonMeta, 3));
      geo.setIndex(new THREE.BufferAttribute(indices, 1));
      return geo;
    }

    const ribbonsGeo = createMergedRibbonsGeometry(5, 72);
    const ribbonsMat = new THREE.ShaderMaterial({
      transparent: true,
      depthWrite: false,
      blending: THREE.AdditiveBlending,
      side: THREE.DoubleSide,
      uniforms: sharedUniforms,
      vertexShader: /* glsl */ `
        attribute vec3 aRibbonMeta; // [normIndex, seed, sideSign]

        uniform float uTime;
        uniform float uPulseAmp;

        varying vec2 vUv;
        varying float vSeed;
        varying float vNormIdx;

        // Cheap 3D axis rotation in vertex shader so ribbons flow independently
        vec3 rotateY(vec3 p, float a) {
          float s = sin(a);
          float c = cos(a);
          return vec3(c * p.x + s * p.z, p.y, -s * p.x + c * p.z);
        }

        vec3 rotateX(vec3 p, float a) {
          float s = sin(a);
          float c = cos(a);
          return vec3(p.x, c * p.y - s * p.z, s * p.y + c * p.z);
        }

        void main() {
          vUv = uv;
          vNormIdx = aRibbonMeta.x;
          vSeed = aRibbonMeta.y;

          float dir = mod(floor(aRibbonMeta.x * 5.0 + 0.5), 2.0) == 0.0 ? 1.0 : -1.0;
          float speed = (0.38 + aRibbonMeta.x * 0.25) * dir;

          // Organic undulation along ribbon length
          float wave = sin(uv.x * 9.0 - uTime * 2.4 + vSeed * 3.0) * (0.08 + uPulseAmp * 0.8);
          vec3 pos = position + normalize(position) * wave;

          // Independent orbital drift per ribbon
          pos = rotateY(pos, uTime * speed + vSeed);
          pos = rotateX(pos, sin(uTime * 0.35 + vSeed) * 0.28);

          gl_Position = projectionMatrix * modelViewMatrix * vec4(pos, 1.0);
        }
      `,
      fragmentShader: /* glsl */ `
        precision mediump float;

        uniform float uTime;
        uniform float uIntensity;
        uniform float uHueShift;
        uniform float uRibbonAlpha;
        uniform float uSurge;

        varying vec2 vUv;
        varying float vSeed;
        varying float vNormIdx;

        void main() {
          // Soft cross-ribbon profile + smooth tip fade
          float crossProfile = sin(vUv.y * 3.14159265);
          float lengthFade = pow(sin(vUv.x * 3.14159265), 0.75);

          // Traveling energy pulse packets along the plasma filament
          float pulse = sin(vUv.x * 12.0 - uTime * 3.6 + vSeed * 4.0) * 0.5 + 0.5;
          pulse = pow(pulse, 2.2);

          vec3 cobalt = vec3(0.03, 0.32, 0.95);
          vec3 cyan   = vec3(0.06, 0.84 + uHueShift * 0.3, 1.0);
          vec3 white  = vec3(0.92, 0.99, 1.0);

          vec3 col = mix(cobalt, cyan, 0.45 + 0.55 * pulse);
          col = mix(col, white, pow(pulse, 2.6) * 0.85);
          col *= uIntensity * (1.0 + uSurge * 0.3);

          float alpha = crossProfile * lengthFade * (0.24 + pulse * 0.62) * uRibbonAlpha;
          gl_FragColor = vec4(col, clamp(alpha, 0.0, 0.9));
        }
      `
    });
    const ribbonsMesh = new THREE.Mesh(ribbonsGeo, ribbonsMat);
    coreRoot.add(ribbonsMesh);

    // ---------------------------------------------------------------------------
    // 8. DRAW CALL 4: 36 ORBITAL ENERGY SPARKS (100% GPU VERTEX ANIMATION)
    // Dynamically scales visible count via uParticleRatio without buffer updates.
    // ---------------------------------------------------------------------------
    const MAX_PARTICLES = 36;
    const particleGeo = new THREE.BufferGeometry();
    const pPositions = new Float32Array(MAX_PARTICLES * 3); // Placeholder origins
    const pOrbitParams = new Float32Array(MAX_PARTICLES * 4); // [radius, theta0, phi0, speed]
    const pVisualParams = new Float32Array(MAX_PARTICLES * 3); // [size, phase, normIndex]

    for (let i = 0; i < MAX_PARTICLES; i++) {
      pOrbitParams[i * 4]     = 1.28 + Math.random() * 1.35; // Orbital radius
      pOrbitParams[i * 4 + 1] = Math.random() * Math.PI * 2; // Initial azimuth
      pOrbitParams[i * 4 + 2] = Math.acos(2 * Math.random() - 1); // Polar angle
      pOrbitParams[i * 4 + 3] = (0.25 + Math.random() * 0.55) * (i % 2 === 0 ? 1 : -1);

      pVisualParams[i * 3]     = 18.0 + Math.random() * 22.0; // Base point size
      pVisualParams[i * 3 + 1] = Math.random() * Math.PI * 2; // Twinkle phase
      pVisualParams[i * 3 + 2] = i / MAX_PARTICLES;           // Normalized index for quality culling
    }

    particleGeo.setAttribute('position', new THREE.BufferAttribute(pPositions, 3));
    particleGeo.setAttribute('aOrbit', new THREE.BufferAttribute(pOrbitParams, 4));
    particleGeo.setAttribute('aVisual', new THREE.BufferAttribute(pVisualParams, 3));

    const particleMat = new THREE.ShaderMaterial({
      transparent: true,
      depthWrite: false,
      blending: THREE.AdditiveBlending,
      uniforms: sharedUniforms,
      vertexShader: /* glsl */ `
        attribute vec4 aOrbit;  // [radius, theta0, phi0, speed]
        attribute vec3 aVisual; // [size, phase, normIndex]

        uniform float uTime;
        uniform float uPixelRatio;
        uniform float uParticleRatio;
        uniform float uPulseAmp;

        varying float vAlpha;

        void main() {
          // Cull excess particles cleanly in vertex shader on Medium/Low profiles
          if (aVisual.z > uParticleRatio) {
            vAlpha = 0.0;
            gl_Position = vec4(2.0, 2.0, 2.0, 1.0);
            gl_PointSize = 0.0;
            return;
          }

          float rad = aOrbit.x + sin(uTime * 1.4 + aVisual.y) * (0.12 + uPulseAmp);
          float theta = aOrbit.y + uTime * aOrbit.w;
          float phi = aOrbit.z + sin(uTime * 0.45 + aVisual.y) * 0.35;

          float sinPhi = sin(phi);
          vec3 pos = vec3(
            rad * sinPhi * cos(theta),
            rad * cos(phi),
            rad * sinPhi * sin(theta)
          );

          vec4 mvPos = modelViewMatrix * vec4(pos, 1.0);
          float twinkle = sin(uTime * 2.6 + aVisual.y * 2.0) * 0.5 + 0.5;
          vAlpha = 0.25 + 0.75 * twinkle;

          gl_PointSize = aVisual.x * uPixelRatio * (1.6 / max(-mvPos.z, 0.5)) * (0.75 + 0.35 * twinkle);
          gl_Position = projectionMatrix * mvPos;
        }
      `,
      fragmentShader: /* glsl */ `
        precision mediump float;

        uniform float uIntensity;
        uniform float uHueShift;
        uniform float uSurge;

        varying float vAlpha;

        void main() {
          if (vAlpha <= 0.01) discard;

          vec2 uv = gl_PointCoord - vec2(0.5);
          float d = length(uv) * 2.0;
          if (d > 1.0) discard;

          // Hot white center + soft cyan radial halo
          float core = exp(-d * d * 8.5);
          float glow = pow(1.0 - d, 2.2) * 0.55;

          vec3 cyan  = vec3(0.06, 0.84 + uHueShift * 0.3, 1.0);
          vec3 white = vec3(0.94, 0.99, 1.0);
          vec3 col = mix(cyan, white, core) * uIntensity * (1.0 + uSurge * 0.35);

          gl_FragColor = vec4(col, (core + glow) * vAlpha);
        }
      `
    });
    const particlesMesh = new THREE.Points(particleGeo, particleMat);
    coreRoot.add(particlesMesh);

    // ---------------------------------------------------------------------------
    // 9. DYNAMIC RESOLUTION & QUALITY GOVERNOR (AUTO / HIGH / MEDIUM / LOW)
    // Monitors rolling frame times over 90 frames on Adreno 618 and adjusts
    // pixelRatio, shader octaves, and active spark count without stutter.
    // ---------------------------------------------------------------------------
    const TIER_ORDER = ['low', 'medium', 'high'];
    const frameTimeWindow = [];
    const WINDOW_SIZE = 90;
    let cooldownFrames = 0;

    function applyQualityTier(tierKey) {
      const cfg = QUALITY_PROFILES[tierKey] || QUALITY_PROFILES.high;
      engineState.activeTier = tierKey;

      const dpr = Math.min(window.devicePixelRatio || 1, cfg.maxPixelRatio);
      renderer.setPixelRatio(dpr);

      sharedUniforms.uPixelRatio.value = dpr;
      sharedUniforms.uDetail.value = cfg.detail;
      sharedUniforms.uParticleRatio.value = cfg.particleRatio;
      sharedUniforms.uRibbonAlpha.value = cfg.ribbonAlpha;

      const prefix = engineState.mode === 'auto' ? 'AUTO•' : '';
      hudProfileEl.textContent = `${prefix}${cfg.name}`;
      hudProfileEl.className =
        tierKey === 'high'
          ? 'text-emerald-300 font-semibold'
          : tierKey === 'medium'
          ? 'text-amber-300 font-semibold'
          : 'text-rose-300 font-semibold';
    }

    function evaluateQualityGovernor(frameMs) {
      if (engineState.mode !== 'auto' || engineState.isIdlePowerSave) return;

      if (cooldownFrames > 0) {
        cooldownFrames--;
        return;
      }

      frameTimeWindow.push(frameMs);
      if (frameTimeWindow.length < WINDOW_SIZE) return;

      let sum = 0;
      for (let i = 0; i < WINDOW_SIZE; i++) sum += frameTimeWindow[i];
      const avgMs = sum / WINDOW_SIZE;
      frameTimeWindow.length = 0;

      const idx = TIER_ORDER.indexOf(engineState.activeTier);

      // Step down if frame time exceeds 19.5ms (< 51 FPS)
      if (avgMs > 19.5 && idx > 0) {
        applyQualityTier(TIER_ORDER[idx - 1]);
        cooldownFrames = 120; // Wait ~2 seconds before re-evaluating
      }
      // Cautiously step up if frame time stays comfortably under 12.5ms (> 80 FPS headroom)
      else if (avgMs < 12.5 && idx < TIER_ORDER.length - 1) {
        applyQualityTier(TIER_ORDER[idx + 1]);
        cooldownFrames = 180;
      }
    }

    // ---------------------------------------------------------------------------
    // 10. TOUCH PARALLAX & BATTERY / LAUNCHER LIFECYCLE MANAGEMENT
    // ---------------------------------------------------------------------------
    function registerUserActivity() {
      engineState.lastInteractionTime = performance.now();
      if (engineState.isIdlePowerSave) {
        engineState.isIdlePowerSave = false;
        hudPowerEl.textContent = 'ACTIVE • 60HZ';
        hudPowerEl.className = 'mono text-[10px] font-semibold tracking-wider uppercase text-cyan-300';
        statusDotEl.className = 'inline-block w-2 h-2 rounded-full bg-cyan-400 shadow-[0_0_8px_#22d3ee]';
        frameTimeWindow.length = 0;
      }
    }

    viewportEl.addEventListener('pointerdown', (e) => {
      registerUserActivity();
      engineState.pointerDown = true;
      engineState.lastPointerX = e.clientX;
      engineState.lastPointerY = e.clientY;
    }, { passive: true });

    window.addEventListener('pointermove', (e) => {
      if (!engineState.pointerDown) return;
      registerUserActivity();
      const dx = e.clientX - engineState.lastPointerX;
      const dy = e.clientY - engineState.lastPointerY;
      engineState.lastPointerX = e.clientX;
      engineState.lastPointerY = e.clientY;

      engineState.targetRotY += dx * 0.0065;
      engineState.targetRotX = Math.max(-0.85, Math.min(0.85, engineState.targetRotX + dy * 0.0065));
    }, { passive: true });

    window.addEventListener('pointerup', () => {
      engineState.pointerDown = false;
    }, { passive: true });

    // Resize handler
    window.addEventListener('resize', () => {
      const w = window.innerWidth;
      const h = window.innerHeight;
      camera.aspect = w / h;
      camera.updateProjectionMatrix();
      renderer.setSize(w, h);
      applyQualityTier(engineState.activeTier);
    });

    // ---------------------------------------------------------------------------
    // 11. MAIN RENDER LOOP WITH IDLE 30FPS THROTTLING & BACKGROUND SUSPENSION
    // ---------------------------------------------------------------------------
    let rafId = null;
    let lastFrameNow = performance.now();
    let idleAccumulator = 0;
    let hudTimer = 0;
    let hudFrames = 0;
    let hudAccumMs = 0;

    function renderLoop(now) {
      if (!engineState.isRunning || engineState.isSuspended) {
        rafId = null;
        return;
      }

      rafId = requestAnimationFrame(renderLoop);

      const rawDtMs = now - lastFrameNow;
      lastFrameNow = now;

      // Clamp huge delta spikes after tab/app resume
      const dtMs = Math.min(rawDtMs, 100);

      // Check 8-second inactivity threshold for Launcher Idle Power-Save (30 FPS)
      if (!engineState.isIdlePowerSave && (now - engineState.lastInteractionTime) > 8000) {
        engineState.isIdlePowerSave = true;
        hudPowerEl.textContent = 'ECO IDLE • 30HZ';
        hudPowerEl.className = 'mono text-[10px] font-semibold tracking-wider uppercase text-amber-300';
        statusDotEl.className = 'inline-block w-2 h-2 rounded-full bg-amber-400 shadow-[0_0_8px_#fbbf24]';
      }

      // Throttle GPU submission to ~30 FPS in Idle Power-Save mode to cut battery & thermal load
      if (engineState.isIdlePowerSave) {
        idleAccumulator += dtMs;
        if (idleAccumulator < 32.5) return;
        idleAccumulator %= 32.5;
      }

      const dtSec = dtMs * 0.001;
      const tempoFactor = engineState.isIdlePowerSave ? 0.65 : 1.0;

      // Smoothly interpolate AI assistant state uniforms
      const targetPreset = AI_STATES[engineState.aiStateKey] || AI_STATES.idle;
      const lerpRate = Math.min(1.0, dtSec * 5.5);

      engineState.currentSpeed += (targetPreset.speed - engineState.currentSpeed) * lerpRate;
      engineState.currentIntensity += (targetPreset.intensity - engineState.currentIntensity) * lerpRate;
      engineState.currentPulseAmp += (targetPreset.pulseAmp - engineState.currentPulseAmp) * lerpRate;
      engineState.currentHueShift += (targetPreset.hueShift - engineState.currentHueShift) * lerpRate;
      engineState.currentScale += (targetPreset.scale - engineState.currentScale) * lerpRate;

      engineState.simTime += dtSec * engineState.currentSpeed * tempoFactor;
      const t = engineState.simTime;

      // Rare stochastic energy flash on inner core
      if (t >= engineState.nextSurgeAt && engineState.surgeValue <= 0.001) {
        engineState.surgeValue = 1.0;
        engineState.nextSurgeAt = t + 3.2 + Math.random() * 4.0;
        sharedUniforms.uSurgeDir.value.set(
          Math.random() * 2 - 1,
          Math.random() * 2 - 1,
          Math.random() * 2 - 1
        ).normalize();
      }
      if (engineState.surgeValue > 0) {
        engineState.surgeValue = Math.max(0, engineState.surgeValue - dtSec * 1.4);
      }

      // Update shared GPU uniforms
      sharedUniforms.uTime.value = t;
      sharedUniforms.uIntensity.value = engineState.currentIntensity;
      sharedUniforms.uPulseAmp.value = engineState.currentPulseAmp;
      sharedUniforms.uHueShift.value = engineState.currentHueShift;
      sharedUniforms.uSurge.value = Math.sin(engineState.surgeValue * Math.PI);

      // Gentle damped touch parallax + organic slow drift
      if (!engineState.pointerDown) {
        engineState.targetRotY += dtSec * 0.14 * tempoFactor;
        engineState.targetRotX *= (1.0 - dtSec * 1.2);
      }
      engineState.currentRotX += (engineState.targetRotX - engineState.currentRotX) * Math.min(1.0, dtSec * 8.0);
      engineState.currentRotY += (engineState.targetRotY - engineState.currentRotY) * Math.min(1.0, dtSec * 8.0);

      coreRoot.rotation.set(engineState.currentRotX, engineState.currentRotY, 0);
      coreRoot.scale.setScalar(engineState.currentScale);

      // Execute the 4 draw calls
      renderer.render(scene, camera);

      // Feed active-mode frame time into Quality Governor
      if (!engineState.isIdlePowerSave) {
        evaluateQualityGovernor(dtMs);
      }

      // Update HUD telemetry every ~400ms to avoid DOM layout thrashing
      hudFrames++;
      hudAccumMs += dtMs;
      hudTimer += dtMs;
      if (hudTimer >= 400) {
        const avgFrameMs = hudAccumMs / hudFrames;
        const fps = Math.min(60, Math.round(1000 / Math.max(avgFrameMs, 1)));
        hudFpsEl.textContent = String(fps);
        hudMsEl.textContent = `${avgFrameMs.toFixed(1)}ms`;
        hudDcEl.textContent = `${renderer.info.render.calls} DC`;
        hudTimer = 0;
        hudFrames = 0;
        hudAccumMs = 0;
      }
    }

    function startLoop() {
      if (rafId === null && engineState.isRunning && !engineState.isSuspended) {
        lastFrameNow = performance.now();
        rafId = requestAnimationFrame(renderLoop);
      }
    }

    function stopLoop() {
      if (rafId !== null) {
        cancelAnimationFrame(rafId);
        rafId = null;
      }
    }

    // Android WebView / Launcher background suspension hooks
    function handleSuspend() {
      engineState.isSuspended = true;
      stopLoop();
    }

    function handleResume() {
      if (document.hidden) return;
      engineState.isSuspended = false;
      registerUserActivity();
      startLoop();
    }

    document.addEventListener('visibilitychange', () => {
      if (document.hidden) handleSuspend();
      else handleResume();
    });
    window.addEventListener('pagehide', handleSuspend);
    window.addEventListener('pageshow', handleResume);
    window.addEventListener('blur', handleSuspend);
    window.addEventListener('focus', handleResume);

    // ---------------------------------------------------------------------------
    // 12. PUBLIC ANDROID WEBVIEW BRIDGE (`window.JarvisCoreAPI`)
    // Callable from Kotlin/Java via webView.evaluateJavascript(...)
    // ---------------------------------------------------------------------------
    function syncUIButtons() {
      document.querySelectorAll('#profile-controls button').forEach((btn) => {
        btn.classList.toggle('active', btn.dataset.profile === engineState.mode);
      });
      document.querySelectorAll('#state-controls button').forEach((btn) => {
        btn.classList.toggle('active', btn.dataset.state === engineState.aiStateKey);
      });
    }

    window.JarvisCoreAPI = {
      setProfile(profile) {
        const key = String(profile || '').toLowerCase();
        registerUserActivity();
        frameTimeWindow.length = 0;
        if (key === 'auto') {
          engineState.mode = 'auto';
          applyQualityTier('high');
        } else if (QUALITY_PROFILES[key]) {
          engineState.mode = key;
          applyQualityTier(key);
        }
        syncUIButtons();
      },

      setState(stateName) {
        const key = String(stateName || '').toLowerCase();
        if (!AI_STATES[key]) return;
        registerUserActivity();
        engineState.aiStateKey = key;
        hudAiStateEl.textContent = `STATE: ${AI_STATES[key].label}`;
        syncUIButtons();
      },

      pause() {
        engineState.isRunning = false;
        stopLoop();
      },

      resume() {
        engineState.isRunning = true;
        engineState.isSuspended = false;
        registerUserActivity();
        startLoop();
      },

      dispose() {
        stopLoop();
        innerCoreGeo.dispose();
        innerCoreMat.dispose();
        outerShellGeo.dispose();
        outerShellMat.dispose();
        ribbonsGeo.dispose();
        ribbonsMat.dispose();
        particleGeo.dispose();
        particleMat.dispose();
        renderer.dispose();
      }
    };

    // Wire bottom mobile HUD controls to JarvisCoreAPI
    document.getElementById('profile-controls').addEventListener('click', (e) => {
      const btn = e.target.closest('button[data-profile]');
      if (btn) window.JarvisCoreAPI.setProfile(btn.dataset.profile);
    });

    document.getElementById('state-controls').addEventListener('click', (e) => {
      const btn = e.target.closest('button[data-state]');
      if (btn) window.JarvisCoreAPI.setState(btn.dataset.state);
    });

    // Boot engine
    applyQualityTier('high');
    startLoop();
  </script>
